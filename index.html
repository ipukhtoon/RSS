function addToCart(id,qty=1,unitType='piece'){
 const p=allProducts.find(x=>x.id===id);if(!p)return;
 const packSize=Math.max(1,Number(p.packSize)||1),isPack=unitType==='pack',stockQty=qty*(isPack?packSize:1);
 const unitPrice=isPack?Number(p.packPrice??(Number(p.price||0)*packSize)):Number(p.price||0);
 const row=cart.find(x=>x.productId===id&&x.unitType===(isPack?'pack':'piece'));
 if(row)row.qty+=qty;else cart.push({productId:id,name:p.name,price:unitPrice,qty,unitType:isPack?'pack':'piece',stockQty,packSize});
 if(row)row.stockQty+=stockQty;renderCart();
}

function renderCart(){
 const host=document.getElementById('cartItems');if(!host)return;
 document.getElementById('cartCount').textContent=cart.reduce((a,x)=>a+(Number(x.stockQty??x.qty)||0),0)+' pcs';
 host.innerHTML=cart.length?cart.map(x=>`<div class="cart-row"><div><strong>${esc(x.name)}</strong><small>${money(x.price)} × ${x.qty} ${x.unitType}</small></div><strong>${money(x.price*x.qty)}</strong></div>`).join(''):'<div class="empty">Cart is empty.</div>';
 updateTotals();
}

function updateTotals(){
 const el=document.getElementById('cartTotals');const btn=document.getElementById('completeSale');if(!el)return;
 const subtotal=cart.reduce((a,x)=>a+x.price*x.qty,0),discount=Math.max(0,Number(document.getElementById('discount')?.value||0)),total=Math.max(0,subtotal-discount),received=Number(document.getElementById('cashReceived')?.value||0);
 const payment=document.getElementById('payment')?.value||'Cash';
 el.innerHTML=`<div class="total-line"><span>Subtotal</span><span>${money(subtotal)}</span></div><div class="total-line"><span>Discount</span><span>− ${money(discount)}</span></div><div class="total-line grand"><span>Total</span><span>${money(total)}</span></div><div class="total-line"><span>Change due</span><strong>${money(Math.max(0,received-total))}</strong></div>`;
 
 if(btn){
   let isValid=true;
   if(cart.length===0)isValid=false;
   else if(payment==='Cash'&&received<=0)isValid=false;
   else if(payment==='Borrow'&&!document.getElementById('borrowCustomer')?.value)isValid=false;
   btn.disabled=!isValid;
 }
}

async function completeSale(){
 if(!cart.length)return;
 const subtotal=cart.reduce((a,x)=>a+x.price*x.qty,0),discount=Math.min(subtotal,Math.max(0,Number(document.getElementById('discount').value||0))),total=subtotal-discount,payment=document.getElementById('payment').value,received=Number(document.getElementById('cashReceived').value||0);
 const customerId=payment==='Borrow'?document.getElementById('borrowCustomer')?.value||'':'';
 const customer=payment==='Borrow'?(await getAll('customers')).find(c=>c.id===customerId):null;
 if(payment==='Borrow'&&!customer){toast('Select a customer account.');return}
 
 const sale={id:uid(),receipt:'RSS-'+Date.now(),date:new Date().toISOString(),items:cart.map(x=>({...x})),subtotal,discount,total,payment,cashReceived:payment==='Borrow'?0:received,change:payment==='Borrow'?0:Math.max(0,received-total),customerId:payment==='Borrow'?customer.id:null,customerName:payment==='Borrow'?customer.name:null,creditOutstanding:payment==='Borrow'?total:0};
 for(const x of cart){const p=allProducts.find(y=>y.id===x.productId);if(p){p.stock-=Number(x.stockQty??x.qty);await put('products',p)}}
 await put('sales',sale);cart=[];toast('Sale completed successfully');showReceipt(sale);render();
}

function calculateCreditBalances(customers,sales,payments){return customers.map(c=>{const total=sales.filter(s=>s.customerId===c.id).reduce((a,s)=>a+Number(s.creditOutstanding??s.total??0),0);const paid=payments.filter(p=>p.customerId===c.id).reduce((a,p)=>a+Number(p.amount||0),0);return {...c,balance:Math.max(0,total-paid)}})}
function renderCreditCustomerCards(list){return list.length?list.map(c=>`<article class="credit-card"><strong>${esc(c.name)}</strong> — Balance: <span class="credit-balance">${money(c.balance)}</span><div class="actions"><button class="secondary" data-credit-history="${esc(c.id)}">View ledger</button></div></article>`).join(''):'<div class="empty">No customers.</div>'}
